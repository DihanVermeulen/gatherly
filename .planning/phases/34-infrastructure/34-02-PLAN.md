    PATH 1 — POST /api/events/:id/participants (around line 432):
    This path currently does: ownership check query → INSERT (no transaction). Replace the entire
    ownership check + INSERT block with a transaction so the count check and INSERT are atomic —
    no concurrent request can insert a 21st participant between the count read and the INSERT.

    Replace the existing ownership check and INSERT with:

    ```typescript
    const ownerCheck = await query(
      "SELECT id, plan_tier FROM events WHERE id = $1 AND organizer_id = $2",
      [id, (req as any).user.userId],
    );
    if (ownerCheck.rows.length === 0) return res.status(403).json({ error: "Forbidden" });
    const planTier = ownerCheck.rows[0].plan_tier || 'free';

    // Wrap count check + INSERT in a transaction to prevent race-window bypass
    const client = await getClient();
    try {
      await client.query("BEGIN");

      if (planTier === 'free') {
        const countResult = await client.query(
          "SELECT COUNT(*)::int AS count FROM participants WHERE event_id = $1",
          [id],
        );
        if (countResult.rows[0].count >= 20) {
          await client.query("ROLLBACK");
          return res.status(403).json({ error: 'participant_cap_reached', limit: 20 });
        }
      }

      const insertResult = await client.query(
        "INSERT INTO participants (event_id, name) VALUES ($1, $2) RETURNING *",
        [id, name],
      );
      await client.query("COMMIT");
      res.status(201).json(insertResult.rows[0]);
    } catch (err) {
      await client.query("ROLLBACK");
      throw err;
    } finally {
      client.release();
    }
    ```

    getClient is already imported at the top of events.ts. Do not add a duplicate import.
    The ownership check (ownerCheck query) remains outside the transaction — it is a read-only
    guard that does not need serialization.
