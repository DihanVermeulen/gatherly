       - Change the response type to `any` (or a union type) and branch on the presence of `participantId`:
         ```typescript
         async redeemMagicLink(token: string): Promise<AuthResponse> {
           const response = await apiClient.post("/api/auth/magic-link/redeem", { token });
           const data = response.data;

           // User-scoped response: existing user detected by backend
           if (data.user.id && !data.user.participantId) {
             return {
               accessToken: data.accessToken,
               user: {
                 id: data.user.id,
                 email: data.user.email,
                 name: data.user.name,
                 role: data.user.role,
                 // eventId hint for navigation (not a JWT claim)
                 eventId: data.user.eventId,
                 eventName: data.user.eventName,
               },
             };
           }

           // Participant-scoped response: no account (existing behavior)
           const { participantId, participantName, eventId, eventName, role } = data.user;
           return {
             accessToken: data.accessToken,
             user: {
               id: participantId,
               email: "",
               name: participantName,
               role,
               participantId,
               eventId,
               eventName,
             },
           };
         }
         ```
       - The key detection logic: if `data.user.id` exists AND `data.user.participantId` does NOT exist, it's a user-scoped response. The participant-scoped response has `participantId` but no `id` field at the top level of user.
