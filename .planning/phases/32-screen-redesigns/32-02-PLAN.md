  **4. Auto-save on toggle (remove Save button):**

  Replace the existing `handleSave` pattern with auto-save on each toggle:

  ```typescript
  const handleToggle = async (type: string, isPremium: boolean, comingSoon?: boolean) => {
    if (comingSoon) return; // Coming Soon modules cannot be toggled
    if (isPremium && isFree) {
      setShowUpgradeModal(true);
      return;
    }

    // Optimistic update
    const newModules = new Set(activeModules);
    if (newModules.has(type)) {
      newModules.delete(type);
    } else {
      newModules.add(type);
    }
    setActiveModules(newModules);

    // Auto-save: send all active modules to API
    try {
      const modulesPayload = Array.from(newModules).map((t) => ({
        type: t,
        status: 'active',
        config: {},
      }));
      await modulesApi.setModules(id, modulesPayload);
    } catch (err: any) {
      // Revert on error
      setActiveModules(activeModules);
      if (err?.response?.data?.error === 'upgrade_required') {
        setShowUpgradeModal(true);
      }
    }
  };
  ```

  Add a `savingModule` state (string | null) to show a spinner on the specific toggle that's saving, and disable other toggles during save to prevent race conditions.
