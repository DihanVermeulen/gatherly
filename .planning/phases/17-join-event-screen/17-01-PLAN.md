    ```typescript
    // Validate an invite code (public endpoint, no auth required)
    validate: (code: string): Promise<InvitePreview> =>
      apiClient
        .post('/api/invites/validate', { code })
        .then((r) => r.data),

    // Accept an invite (public endpoint, but app enforces auth client-side)
    accept: (code: string, participantName: string, email?: string): Promise<JoinResult> =>
      apiClient
        .post(`/api/invites/${code}/accept`, { participantName, email })
        .then((r) => r.data),
    ```

    2. Create `apps/gatherly-mobile/app/utils/pendingInvite.ts` — a module-level variable utility for preserving the invite code across auth navigation. This is NOT AsyncStorage — it's a simple module variable that persists within the same JS session (survives navigation stack changes, does not survive app restart, which is fine since auth screens are in the same session).

    ```typescript
    // Module-level variable — persists across component remounts within same session
    let _pendingInviteCode: string | null = null;

    export function setPendingInviteCode(code: string): void {
      _pendingInviteCode = code;
    }

    export function consumePendingInviteCode(): string | null {
      const code = _pendingInviteCode;
      _pendingInviteCode = null;
      return code;
    }
    ```

    Create the `utils/` directory if it does not exist.
