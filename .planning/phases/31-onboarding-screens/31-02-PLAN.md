    **handleComplete function:**
    ```typescript
    const handleComplete = async () => {
      setLoading(true);
      try {
        await usersApi.updateMe({ interests: selectedIds, onboardingComplete: true });
        await updateUser({ onboardingComplete: true });
      } catch (err) {
        // Non-fatal: still complete onboarding locally
        console.error('Failed to save preferences:', err);
        await updateUser({ onboardingComplete: true });
      } finally {
        setLoading(false);
      }
    };
    ```
    Where `selectedIds` is the array of selected category `id` strings. `updateUser` comes from `useSession()`.

    **handleSkip function:**
    Same as handleComplete but without sending interests:
    ```typescript
    const handleSkip = async () => {
      setLoading(true);
      try {
        await usersApi.updateMe({ onboardingComplete: true });
        await updateUser({ onboardingComplete: true });
      } catch {
        await updateUser({ onboardingComplete: true });
      } finally {
        setLoading(false);
      }
    };
    ```

    **CRITICAL:** After `updateUser({ onboardingComplete: true })`, the Stack.Protected guard in _layout.tsx re-evaluates automatically. The guard `!!session && !user?.participantId && !user?.onboardingComplete` becomes false, and the user is routed to the main app tabs. Do NOT manually `router.replace('/(tabs)')` — let the guard handle navigation.
